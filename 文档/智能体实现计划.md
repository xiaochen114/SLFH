      def 调用(self, system_prompt: str, user_prompt: str, tools=None, max_tokens=2048) -> dict:
          """返回 {"content": "...", "tool_calls": [...]} 或 {"error": "..."}"""
          if not self._key and self._provider != "ollama":
              return {"error": "未配置API密钥，请设环境变量 DEEPSEEK_API_KEY"}
          # POST {base_url}/chat/completions
          # 超时30秒，异常返回error

### 任务4: 创建 智能体.py — 核心Agent

**文件：** `中央大脑/智能体.py`

- [ ] **步骤1：核心类骨架**
  ```python
  class 智能体:
      def __init__(self, 注册中心, 通信, llm后端, 感知主机=None, patrol=None):
          self._registry = 注册中心
          self._comm = 通信
          self._llm = llm后端
          self._tools = 智能体工具(注册中心, 通信)
          self._comm_mgr = 通信管理器智能()
          self._ph = 感知主机
          self._patrol = patrol
          self._running = False
          self._对话历史 = []  # 保留最近10轮

      def 启动(self):
          """启后台循环线程"""

      def 停止(self):
          self._running = False

      def _循环(self):
          """每2秒: 采集→判断通信→调LLM→执行"""

      def _决策(self, 状态: dict) -> dict:
          """调LLM，返回结构化决策"""

      def _执行(self, 决策: dict):
          """分类执行: 调度/适配/修复/git/问用户"""

      def _紧急拦截(self, 状态: dict) -> bool:
          """火情L3→急停，返回True表示已拦截"""
  ```

- [ ] **步骤2：紧急硬编码**
  ```python
  def _紧急拦截(self, 状态):
      events = 状态.get("events", [])
      for ev in events:
          if ev.get("type") == "fire_detected":
              rid = ev.get("robot_id", "")
              self._tools.发送指令(rid, "alert", {"reason": "fire"})
              print(f"[智能体] 紧急急停 {rid}")
              return True
      return False
  ```
